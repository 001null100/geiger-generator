cmake_minimum_required(VERSION 3.24)
project(GeigerGenerator VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Every static dependency linked into the CLAP module must be position-independent.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(FetchContent)
option(GEIGER_BUILD_PLUGIN "Build the CLAP and native GUI" ON)
add_library(geiger_dsp STATIC Source/Engine.cpp)
target_include_directories(geiger_dsp PUBLIC Source)
set_target_properties(geiger_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(BUILD_TESTING)
  add_executable(GeigerEngineTests tests/EngineTests.cpp)
  target_link_libraries(GeigerEngineTests PRIVATE geiger_dsp)
  add_test(NAME geiger_dsp COMMAND GeigerEngineTests)
endif()
add_executable(GeigerRender tools/Render.cpp)
target_link_libraries(GeigerRender PRIVATE geiger_dsp)
if(GEIGER_BUILD_PLUGIN)
  set(NULLCLAP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(NULLCLAP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(null_clap GIT_REPOSITORY https://github.com/001null100/null-clap.git GIT_TAG 8612a19fbe7fbe81b249c22e1593da18bb998b97 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(null_clap)
  if(NOT EXISTS "${null_clap_SOURCE_DIR}/include/nullclap/PhysicalPixelGuiSizing.hpp")
    message(FATAL_ERROR "The pinned null-clap checkout is incomplete: missing PhysicalPixelGuiSizing.hpp")
  endif()
  FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(juce)
  # JUCE supplies drawing, controls and native windows, never audio/CLAP wrappers.
  add_library(geiger_ui STATIC Source/Plugin.cpp Source/Gui.cpp Source/Editor.cpp)
  target_include_directories(geiger_ui PUBLIC Source "${juce_SOURCE_DIR}/modules" "${null_clap_SOURCE_DIR}/include")
  target_link_libraries(geiger_ui PUBLIC nullclap::nullclap geiger_dsp PRIVATE juce::juce_gui_basics juce::juce_recommended_config_flags)
  target_compile_definitions(geiger_ui PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  set_target_properties(geiger_ui PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  nullclap_add_plugin(GeigerGenerator OUTPUT_NAME GeigerGenerator SOURCES Source/Entry.cpp)
  target_link_libraries(GeigerGenerator PRIVATE geiger_ui)
  if(BUILD_TESTING)
    add_executable(GeigerHostTests tests/HostTests.cpp)
    target_link_libraries(GeigerHostTests PRIVATE geiger_ui)
    add_test(NAME geiger_host COMMAND GeigerHostTests)
    add_executable(GeigerUiSmoke tests/UiSmoke.cpp)
    target_link_libraries(GeigerUiSmoke PRIVATE geiger_ui)
  endif()
  if(MSVC)
    target_compile_options(geiger_ui PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(geiger_ui PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
